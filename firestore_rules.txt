rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }

    function staffProfile() {
      return get(
        /databases/$(database)/documents/users/$(request.auth.uid)
      ).data;
    }

    function isStaff() {
      return isSignedIn()
        && exists(
          /databases/$(database)/documents/users/$(request.auth.uid)
        )
        && staffProfile().isActive == true
        && staffProfile().role in ['operator', 'supervisor', 'admin'];
    }

    function isAdmin() {
      return isSignedIn()
        && exists(
          /databases/$(database)/documents/users/$(request.auth.uid)
        )
        && staffProfile().isActive == true
        && staffProfile().role == 'admin';
    }

    function isAgent() {
      return isSignedIn()
        && exists(
          /databases/$(database)/documents/users/$(request.auth.uid)
        )
        && staffProfile().isActive == true
        && staffProfile().role == 'agent';
    }

    function isOrderOwner() {
      return isSignedIn()
        && resource.data.customerAuthUid == request.auth.uid;
    }

    function hasValidOrderKeys() {
      return request.resource.data.keys().hasOnly([
        'schemaVersion',
        'reference',
        'source',
        'customerAuthUid',
        'clientName',
        'clientWhatsappPhone',
        'service',
        'network',
        'operationType',
        'offerId',
        'offerLabel',
        'isCustomOffer',
        'amount',
        'beneficiaryPhone',
        'status',
        'paymentStatus',
        'createdAt',
        'updatedAt',
        'paymentRequestSentAt',
        'paymentDeclaredAt',
        'paymentPayerName',
        'paymentPayerPhone',
        'paymentApproximateTime',
        'paymentDeclaredReference',
        'expiresAt',
        'expiredAt',
        'paymentConfirmedAt',
        'paidAt',
        'paymentReference',
        'originalWhatsappMessage',
        'internalNotes',
        'takenByUserId',
        'takenAt',
        'completedAt',
        'failureReason',
        'observation',
        'customerConfirmationStatus',
        'customerConfirmationCompletedAt',
        'assignedAgentId',
        'assignedAgentName',
        'assignedByUserId',
        'assignedAt',
        'assignmentMode',
        'assignmentStatus'
      ]);
    }


    function hasValidOrderAssignmentValues() {
      return (
          request.resource.data.get('assignedAgentId', null) == null
          || request.resource.data.get('assignedAgentId', null) is string
        )
        && (
          request.resource.data.get('assignedAgentName', null) == null
          || request.resource.data.get('assignedAgentName', null) is string
        )
        && (
          request.resource.data.get('assignedByUserId', null) == null
          || request.resource.data.get('assignedByUserId', null) is string
        )
        && (
          request.resource.data.get('assignedAt', null) == null
          || request.resource.data.get('assignedAt', null) is timestamp
        )
        && (
          request.resource.data.get('assignmentMode', null) == null
          || request.resource.data.get('assignmentMode', null) in [
            'manual',
            'automatic'
          ]
        )
        && request.resource.data.get('assignmentStatus', 'unassigned') in [
          'unassigned',
          'assigned',
          'accepted',
          'refused'
        ];
    }

    function hasValidCommonOrderValues() {
      return hasValidOrderKeys()
        && hasValidOrderAssignmentValues()
        && request.resource.data.schemaVersion == 1
        && request.resource.data.reference is string
        && request.resource.data.reference.size() >= 12
        && request.resource.data.clientName is string
        && request.resource.data.clientName.size() >= 2
        && request.resource.data.clientName.size() <= 80
        && request.resource.data.clientWhatsappPhone is string
        && request.resource.data['service'] in [
          'unitTransfer',
          'internetSubscription',
          'calls',
          'other'
        ]
        && request.resource.data.network in ['orange', 'mtn', 'moov']
        && request.resource.data.operationType in [
          'unitTransfer',
          'internetSubscription',
          'callBundle',
          'mixedBundle',
          'other'
        ]
        && (
          request.resource.data.offerId == null
          || request.resource.data.offerId is string
        )
        && request.resource.data.offerLabel is string
        && request.resource.data.offerLabel.size() >= 2
        && request.resource.data.offerLabel.size() <= 200
        && request.resource.data.isCustomOffer is bool
        && request.resource.data.amount is int
        && request.resource.data.amount > 0
        && request.resource.data.amount <= 1000000
        && request.resource.data.beneficiaryPhone is string
        && request.resource.data.expiresAt is timestamp
        && (
          request.resource.data.get('expiredAt', null) == null
          || request.resource.data.get('expiredAt', null) is timestamp
        )
        && request.resource.data.status in [
          'awaitingPayment',
          'paymentToVerify',
          'paidReady',
          'inProgress',
          'onHold',
          'awaitingCustomerConfirmation',
          'completed',
          'failed',
          'expired',
          'cancelled',
          'refundPending',
          'refunded'
        ]
        && request.resource.data.paymentStatus in [
          'notDeclared',
          'pending',
          'declared',
          'confirmed',
          'rejected',
          'expired'
        ]
        && request.resource.data.customerConfirmationStatus in [
          'pending',
          'sent',
          'skipped'
        ];
    }

    function isValidOfferSnapshotAtCreation() {
      let offerId = request.resource.data.offerId;

      return (
        request.resource.data.isCustomOffer == true
        && offerId == null
      ) || (
        request.resource.data.isCustomOffer == false
        && request.resource.data['service'] == 'unitTransfer'
        && offerId == null
      ) || (
        request.resource.data.isCustomOffer == false
        && request.resource.data['service'] in ['internetSubscription', 'calls']
        && offerId is string
        && exists(
          /databases/$(database)/documents/offers/$(offerId)
        )
        && get(
          /databases/$(database)/documents/offers/$(offerId)
        ).data.isActive == true
        && get(
          /databases/$(database)/documents/offers/$(offerId)
        ).data.network == request.resource.data.network
        && get(
          /databases/$(database)/documents/offers/$(offerId)
        ).data['service'] == request.resource.data['service']
        && get(
          /databases/$(database)/documents/offers/$(offerId)
        ).data.operationType == request.resource.data.operationType
        && get(
          /databases/$(database)/documents/offers/$(offerId)
        ).data.catalogLabel == request.resource.data.offerLabel
        && get(
          /databases/$(database)/documents/offers/$(offerId)
        ).data.sellingPrice == request.resource.data.amount
      );
    }

    function isValidCustomerOrderCreation() {
      return isSignedIn()
        && hasValidCommonOrderValues()
        && isValidOfferSnapshotAtCreation()
        && request.resource.data.source == 'customerWeb'
        && request.resource.data.customerAuthUid == request.auth.uid
        && request.resource.data['service'] in [
          'unitTransfer',
          'internetSubscription',
          'calls'
        ]
        && request.resource.data.operationType in [
          'unitTransfer',
          'internetSubscription',
          'callBundle',
          'mixedBundle'
        ]
        && request.resource.data.status == 'awaitingPayment'
        && request.resource.data.paymentStatus == 'notDeclared'
        && request.resource.data.createdAt == request.time
        && request.resource.data.updatedAt == request.time
        && request.resource.data.paymentRequestSentAt == null
        && request.resource.data.paymentDeclaredAt == null
        && request.resource.data.paymentPayerName == null
        && request.resource.data.paymentPayerPhone == null
        && request.resource.data.paymentApproximateTime == null
        && request.resource.data.paymentDeclaredReference == null
        && request.resource.data.get('expiredAt', null) == null
        && request.resource.data.paymentConfirmedAt == null
        && request.resource.data.paidAt == null
        && request.resource.data.paymentReference == null
        && request.resource.data.takenByUserId == null
        && request.resource.data.takenAt == null
        && request.resource.data.completedAt == null
        && request.resource.data.failureReason == null
        && request.resource.data.observation == null
        && request.resource.data.customerConfirmationStatus == 'pending'
        && request.resource.data.customerConfirmationCompletedAt == null
        && request.resource.data.get('assignedAgentId', null) == null
        && request.resource.data.get('assignedAgentName', null) == null
        && request.resource.data.get('assignedByUserId', null) == null
        && request.resource.data.get('assignedAt', null) == null
        && request.resource.data.get('assignmentMode', null) == null
        && request.resource.data.get('assignmentStatus', 'unassigned') == 'unassigned';
    }

    function isValidStaffOfferReferenceAtCreation() {
      let offerId = request.resource.data.offerId;

      return request.resource.data.isCustomOffer == true
        || request.resource.data['service'] == 'unitTransfer'
        || (
          request.resource.data.isCustomOffer == false
          && offerId is string
          && exists(
            /databases/$(database)/documents/offers/$(offerId)
          )
          && get(
            /databases/$(database)/documents/offers/$(offerId)
          ).data.isActive == true
        );
    }

    function isValidStaffOrderCreation() {
      return isStaff()
        && hasValidCommonOrderValues()
        && isValidStaffOfferReferenceAtCreation()
        && request.resource.data.source == 'operatorApp'
        && request.resource.data.customerAuthUid == null
        && request.resource.data.status == 'awaitingPayment'
        && request.resource.data.paymentStatus == 'pending'
        && request.resource.data.createdAt == request.time
        && request.resource.data.updatedAt == request.time
        && request.resource.data.paymentRequestSentAt == null
        && request.resource.data.paymentDeclaredAt == null
        && request.resource.data.paymentPayerName == null
        && request.resource.data.paymentPayerPhone == null
        && request.resource.data.paymentApproximateTime == null
        && request.resource.data.paymentDeclaredReference == null
        && request.resource.data.get('expiredAt', null) == null
        && request.resource.data.paymentConfirmedAt == null
        && request.resource.data.paidAt == null
        && request.resource.data.paymentReference == null
        && request.resource.data.takenByUserId == null
        && request.resource.data.takenAt == null
        && request.resource.data.completedAt == null
        && request.resource.data.failureReason == null
        && request.resource.data.observation == null
        && request.resource.data.customerConfirmationStatus == 'pending'
        && request.resource.data.customerConfirmationCompletedAt == null
        && request.resource.data.get('assignedAgentId', null) == null
        && request.resource.data.get('assignedAgentName', null) == null
        && request.resource.data.get('assignedByUserId', null) == null
        && request.resource.data.get('assignedAt', null) == null
        && request.resource.data.get('assignmentMode', null) == null
        && request.resource.data.get('assignmentStatus', 'unassigned') == 'unassigned';
    }

    function immutableCoreOrderFieldsAreUnchanged() {
      return request.resource.data.schemaVersion == resource.data.schemaVersion
        && request.resource.data.reference == resource.data.reference
        && request.resource.data.source == resource.data.source
        && request.resource.data.customerAuthUid == resource.data.customerAuthUid
        && request.resource.data.clientName == resource.data.clientName
        && request.resource.data.clientWhatsappPhone == resource.data.clientWhatsappPhone
        && request.resource.data['service'] == resource.data['service']
        && request.resource.data.network == resource.data.network
        && request.resource.data.operationType == resource.data.operationType
        && request.resource.data.offerId == resource.data.offerId
        && request.resource.data.offerLabel == resource.data.offerLabel
        && request.resource.data.isCustomOffer == resource.data.isCustomOffer
        && request.resource.data.amount == resource.data.amount
        && request.resource.data.beneficiaryPhone == resource.data.beneficiaryPhone
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.expiresAt == resource.data.expiresAt
        && request.resource.data.originalWhatsappMessage == resource.data.originalWhatsappMessage;
    }

    function onlyStaffMutableFieldsChanged() {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly([
        'updatedAt',
        'status',
        'paymentStatus',
        'paymentRequestSentAt',
        'expiredAt',
        'paymentConfirmedAt',
        'paidAt',
        'paymentReference',
        'internalNotes',
        'takenByUserId',
        'takenAt',
        'completedAt',
        'failureReason',
        'observation',
        'customerConfirmationStatus',
        'customerConfirmationCompletedAt',
        'assignedAgentId',
        'assignedAgentName',
        'assignedByUserId',
        'assignedAt',
        'assignmentMode',
        'assignmentStatus'
      ]);
    }

    function isValidStaffOrderUpdate() {
      return isStaff()
        && hasValidCommonOrderValues()
        && immutableCoreOrderFieldsAreUnchanged()
        && onlyStaffMutableFieldsChanged()
        && request.resource.data.updatedAt == request.time;
    }


    function targetAgentProfile(agentId) {
      return get(
        /databases/$(database)/documents/agentProfiles/$(agentId)
      ).data;
    }

    function targetAgentUser(agentId) {
      return get(
        /databases/$(database)/documents/users/$(agentId)
      ).data;
    }

    function targetAgentHasEnoughCapacity(agentId) {
      let profile = targetAgentProfile(agentId);
      let network = request.resource.data.network;
      let amount = request.resource.data.amount;

      return (
          network == 'orange'
          && profile.orangeCapacity >= amount
        ) || (
          network == 'mtn'
          && profile.mtnCapacity >= amount
        ) || (
          network == 'moov'
          && profile.moovCapacity >= amount
        );
    }

    function isValidAdminManualAssignment() {
      let agentId = request.resource.data.get('assignedAgentId', null);

      return isAdmin()
        && agentId is string
        && hasValidCommonOrderValues()
        && immutableCoreOrderFieldsAreUnchanged()
        && resource.data.status == 'paidReady'
        && resource.data.paymentStatus == 'confirmed'
        && request.resource.data.status == resource.data.status
        && request.resource.data.paymentStatus == resource.data.paymentStatus
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'updatedAt',
          'assignedAgentId',
          'assignedAgentName',
          'assignedByUserId',
          'assignedAt',
          'assignmentMode',
          'assignmentStatus'
        ])
        && exists(
          /databases/$(database)/documents/users/$(agentId)
        )
        && exists(
          /databases/$(database)/documents/agentProfiles/$(agentId)
        )
        && targetAgentUser(agentId).role == 'agent'
        && targetAgentUser(agentId).isActive == true
        && targetAgentProfile(agentId).availability == 'available'
        && request.resource.data.network in targetAgentProfile(agentId).authorizedNetworks
        && request.resource.data.network in targetAgentProfile(agentId).activeNetworks
        && targetAgentHasEnoughCapacity(agentId)
        && request.resource.data.assignedAgentName is string
        && request.resource.data.assignedAgentName == targetAgentUser(agentId).name
        && request.resource.data.assignedAgentName.size() >= 2
        && request.resource.data.assignedAgentName.size() <= 80
        && request.resource.data.assignedByUserId == request.auth.uid
        && request.resource.data.assignedAt == request.time
        && request.resource.data.assignmentMode == 'manual'
        && request.resource.data.assignmentStatus == 'assigned'
        && request.resource.data.updatedAt == request.time;
    }

    function isValidCustomerPaymentDeclaration() {
      let wasExpired = resource.data.status == 'expired';
      let isLate = request.time >= resource.data.expiresAt;
      let oldExpiredAt = resource.data.get('expiredAt', null);
      let newExpiredAt = request.resource.data.get('expiredAt', null);

      return isOrderOwner()
        && hasValidCommonOrderValues()
        && immutableCoreOrderFieldsAreUnchanged()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'updatedAt',
          'status',
          'paymentStatus',
          'paymentDeclaredAt',
          'paymentPayerName',
          'paymentPayerPhone',
          'paymentApproximateTime',
          'paymentDeclaredReference',
          'expiredAt'
        ])
        && resource.data.source == 'customerWeb'
        && resource.data.status in ['awaitingPayment', 'expired']
        && resource.data.paymentStatus in ['notDeclared', 'expired']
        && resource.data.paymentDeclaredAt == null
        && request.resource.data.status == (
          wasExpired || isLate ? 'expired' : 'paymentToVerify'
        )
        && request.resource.data.paymentStatus == 'declared'
        && request.resource.data.paymentDeclaredAt == request.time
        && request.resource.data.paymentPayerName is string
        && request.resource.data.paymentPayerName.size() >= 2
        && request.resource.data.paymentPayerName.size() <= 80
        && request.resource.data.paymentPayerPhone is string
        && request.resource.data.paymentPayerPhone.matches('^[+]225(01|05|07)[0-9]{8}$')
        && request.resource.data.paymentApproximateTime is string
        && request.resource.data.paymentApproximateTime.matches('^([01][0-9]|2[0-3]):[0-5][0-9]$')
        && (
          request.resource.data.paymentDeclaredReference == null
          || (
            request.resource.data.paymentDeclaredReference is string
            && request.resource.data.paymentDeclaredReference.size() <= 80
          )
        )
        && (
          (!wasExpired && !isLate && newExpiredAt == oldExpiredAt)
          || (
            (wasExpired || isLate)
            && (
              (oldExpiredAt == null && newExpiredAt == request.time)
              || (oldExpiredAt != null && newExpiredAt == oldExpiredAt)
            )
          )
        )
        && request.resource.data.updatedAt == request.time;
    }

    function isValidCustomerExpiration() {
      return isOrderOwner()
        && hasValidCommonOrderValues()
        && immutableCoreOrderFieldsAreUnchanged()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'updatedAt',
          'status',
          'paymentStatus',
          'expiredAt'
        ])
        && resource.data.source == 'customerWeb'
        && resource.data.status in ['awaitingPayment', 'paymentToVerify']
        && resource.data.paymentStatus in [
          'notDeclared',
          'pending',
          'declared',
          'expired'
        ]
        && resource.data.paymentStatus != 'confirmed'
        && request.time >= resource.data.expiresAt
        && request.resource.data.status == 'expired'
        && request.resource.data.paymentStatus == (
          resource.data.paymentStatus == 'declared' ? 'declared' : 'expired'
        )
        && request.resource.data.expiredAt == request.time
        && request.resource.data.updatedAt == request.time;
    }

    function isValidPendingStaffProfileCreation(userId) {
      return isSignedIn()
        && request.auth.uid == userId
        && request.auth.token.email is string
        && request.resource.data.keys().hasOnly([
          'schemaVersion',
          'name',
          'email',
          'phoneNumber',
          'role',
          'isActive',
          'createdAt',
          'updatedAt'
        ])
        && request.resource.data.schemaVersion == 1
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 2
        && request.resource.data.name.size() <= 80
        && request.resource.data.email is string
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.phoneNumber is string
        && request.resource.data.phoneNumber == ''
        && request.resource.data.role == 'pending'
        && request.resource.data.isActive == false
        && request.resource.data.createdAt == request.time
        && request.resource.data.updatedAt == request.time;
    }

    function hasValidOfferKeys() {
      return request.resource.data.keys().hasOnly([
        'schemaVersion',
        'network',
        'service',
        'operationType',
        'title',
        'catalogLabel',
        'description',
        'sellingPrice',
        'details',
        'badgeLabel',
        'category',
        'validity',
        'volume',
        'minutes',
        'sms',
        'eligibility',
        'isActive',
        'displayOrder',
        'createdAt',
        'updatedAt'
      ]);
    }

    function hasValidOfferValues() {
      return hasValidOfferKeys()
        && request.resource.data.schemaVersion == 1
        && request.resource.data.network in ['orange', 'mtn', 'moov']
        && request.resource.data['service'] in [
          'internetSubscription',
          'calls'
        ]
        && request.resource.data.operationType in [
          'internetSubscription',

          'mixedBundle'
        ]
        && request.resource.data.title is string
        && request.resource.data.title.size() >= 2
        && request.resource.data.title.size() <= 100
        && request.resource.data.catalogLabel is string
        && request.resource.data.catalogLabel.size() >= 2
        && request.resource.data.catalogLabel.size() <= 200

        )

    function isValidOfferCreation() {
      return isAdmin() && hasValidOfferValues();
    }

    function isValidOfferUpdate() {
      return isAdmin()
        && hasValidOfferValues()
        && request.resource.data.createdAt == resource.data.createdAt;
    }

    function hasValidAgentProfileKeys() {
      return request.resource.data.keys().hasOnly([
        'schemaVersion',
        'userId',
        'agentCode',
        'availability',
        'zoneIds',
        'authorizedNetworks',
        'activeNetworks',

        'dailyTransactionLimit',

        'lastCapacityUpdateAt',

      ]);
    }

        && request.resource.data.schemaVersion == 1

        && request.resource.data.agentCode is string
        && request.resource.data.agentCode.size() >= 3
        && request.resource.data.agentCode.size() <= 30
        && request.resource.data.availability in [
          'available',
          'unavailable'
        ]
        && request.resource.data.zoneIds is list
        && request.resource.data.zoneIds.size() <= 20
        && request.resource.data.authorizedNetworks is list
        && request.resource.data.authorizedNetworks.size() <= 3
        && request.resource.data.authorizedNetworks.hasOnly([
          'orange',
          'mtn',
          'moov'
        ])
        && request.resource.data.activeNetworks is list
        && request.resource.data.activeNetworks.size() <= 3
        && request.resource.data.activeNetworks.hasOnly([
          'orange',
          'mtn',
          'moov'
        ])
        && request.resource.data.activeNetworks.hasOnly(
          request.resource.data.authorizedNetworks
        )
        && request.resource.data.orangeCapacity is int
        && request.resource.data.orangeCapacity >= 0
        && request.resource.data.orangeCapacity <= 100000000
        && request.resource.data.mtnCapacity is int
        && request.resource.data.mtnCapacity >= 0
        && request.resource.data.mtnCapacity <= 100000000
        && request.resource.data.moovCapacity is int
        && request.resource.data.moovCapacity >= 0
        && request.resource.data.moovCapacity <= 100000000
        && request.resource.data.dailyTransactionLimit is int
        && request.resource.data.dailyTransactionLimit >= 0
        && request.resource.data.dailyTransactionLimit <= 100000000
        && request.resource.data.maxTransactionsPerDay is int
        && request.resource.data.maxTransactionsPerDay >= 0
        && request.resource.data.maxTransactionsPerDay <= 10000
        && (
          request.resource.data.lastCapacityUpdateAt == null
          || request.resource.data.lastCapacityUpdateAt is timestamp
        )
        && (
          request.resource.data.lastSeenAt == null
          || request.resource.data.lastSeenAt is timestamp
        )
        && request.resource.data.createdAt is timestamp
        && request.resource.data.updatedAt is timestamp;
    }

    function isValidAdminAgentProfileCreation(agentId) {
      return isAdmin()
        && getAfter(
          /databases/$(database)/documents/users/$(agentId)
        ).data.role == 'agent'
        && getAfter(
          /databases/$(database)/documents/users/$(agentId)
        ).data.isActive == true
        && hasValidAgentProfileValues(agentId)
        && request.resource.data.createdAt == request.time
        && request.resource.data.updatedAt == request.time;
    }

    function isValidAdminAgentProfileUpdate(agentId) {
      return isAdmin()
        && hasValidAgentProfileValues(agentId)
        && request.resource.data.userId == resource.data.userId
        && request.resource.data.agentCode == resource.data.agentCode
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.updatedAt == request.time;
    }

    function isValidAgentOperationalUpdate(agentId) {
      return isAgent()
        && request.auth.uid == agentId
        && hasValidAgentProfileValues(agentId)
        && request.resource.data.userId == resource.data.userId
        && request.resource.data.agentCode == resource.data.agentCode
        && request.resource.data.zoneIds == resource.data.zoneIds
        && request.resource.data.authorizedNetworks == resource.data.authorizedNetworks
        && request.resource.data.dailyTransactionLimit == resource.data.dailyTransactionLimit
        && request.resource.data.maxTransactionsPerDay == resource.data.maxTransactionsPerDay
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'availability',
          'activeNetworks',
          'orangeCapacity',
          'mtnCapacity',
          'moovCapacity',
          'lastCapacityUpdateAt',
          'lastSeenAt',
          'updatedAt'
        ])
        && request.resource.data.updatedAt == request.time;
    }

    function hasValidZoneKeys() {
      return request.resource.data.keys().hasOnly([
        'schemaVersion',
        'name',
        'city',
        'region',
        'isActive',
        'createdAt',
        'updatedAt'
      ]);
    }

    function hasValidZoneValues() {
      return hasValidZoneKeys()
        && request.resource.data.schemaVersion == 1
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 2
        && request.resource.data.name.size() <= 100
        && request.resource.data.city is string
        && request.resource.data.city.size() >= 2
        && request.resource.data.city.size() <= 100
        && request.resource.data.region is string
        && request.resource.data.region.size() >= 2
        && request.resource.data.region.size() <= 100
        && request.resource.data.isActive is bool
        && request.resource.data.createdAt is timestamp
        && request.resource.data.updatedAt is timestamp;
    }

    function isValidZoneCreation() {
      return isAdmin()
        && hasValidZoneValues()
        && request.resource.data.createdAt == request.time
        && request.resource.data.updatedAt == request.time;
    }

    function isValidZoneUpdate() {
      return isAdmin()
        && hasValidZoneValues()
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.updatedAt == request.time;
    }

    function hasValidAgentIssueKeys() {
      return request.resource.data.keys().hasOnly([
        'schemaVersion',
        'agentId',
        'type',
        'network',
        'description',
        'status',
        'createdAt',
        'updatedAt',
        'resolvedAt',
        'resolvedBy'
      ]);
    }

    function hasValidAgentIssueValues() {
      return hasValidAgentIssueKeys()
        && request.resource.data.schemaVersion == 1
        && request.resource.data.agentId is string
        && request.resource.data.type in [
          'network',
          'balance',
          'technical',
          'other'
        ]
        && (
          request.resource.data.network == null
          || request.resource.data.network in ['orange', 'mtn', 'moov']
        )
        && request.resource.data.description is string
        && request.resource.data.description.size() >= 3
        && request.resource.data.description.size() <= 1000
        && request.resource.data.status in [
          'open',
          'acknowledged',
          'resolved',
          'cancelled'
        ]
        && request.resource.data.createdAt is timestamp
        && request.resource.data.updatedAt is timestamp
        && (
          request.resource.data.resolvedAt == null
          || request.resource.data.resolvedAt is timestamp
        )
        && (
          request.resource.data.resolvedBy == null
          || request.resource.data.resolvedBy is string
        );
    }

    function isValidAgentIssueCreation() {
      return isAgent()
        && request.resource.data.agentId == request.auth.uid
        && hasValidAgentIssueValues()
        && request.resource.data.status == 'open'
        && request.resource.data.createdAt == request.time
        && request.resource.data.updatedAt == request.time
        && request.resource.data.resolvedAt == null
        && request.resource.data.resolvedBy == null;
    }

    function isValidAdminAgentIssueUpdate() {
      return isAdmin()
        && hasValidAgentIssueValues()
        && request.resource.data.agentId == resource.data.agentId
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.updatedAt == request.time;
    }

    function hasValidStaffUserValues() {
      return request.resource.data.keys().hasOnly([
        'schemaVersion',
        'name',
        'email',
        'phoneNumber',
        'role',
        'isActive',
        'createdAt',
        'updatedAt'
      ])
        && request.resource.data.schemaVersion == 1
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 2
        && request.resource.data.name.size() <= 80
        && request.resource.data.email is string
        && request.resource.data.phoneNumber is string
        && request.resource.data.phoneNumber.size() <= 30
        && request.resource.data.role in [
          'pending',
          'operator',
          'supervisor',
          'agent',
          'admin'
        ]
        && request.resource.data.isActive is bool
        && request.resource.data.createdAt is timestamp
        && request.resource.data.updatedAt is timestamp;
    }

    function isValidAdminUserUpdate(userId) {
      return isAdmin()
        && hasValidStaffUserValues()
        && request.resource.data.email == resource.data.email
        && request.resource.data.createdAt == resource.data.createdAt
        && request.resource.data.updatedAt == request.time
        && (
          request.auth.uid != userId
          || (
            request.resource.data.role == resource.data.role
            && request.resource.data.isActive == resource.data.isActive
          )
        );
    }


    function hasValidOrderAssignmentEventKeys() {
      return request.resource.data.keys().hasOnly([
        'schemaVersion',
        'orderId',
        'orderReference',
        'agentId',
        'agentName',
        'assignedByUserId',
        'mode',
        'status',
        'assignedAt',
        'acceptedAt',
        'refusedAt',
        'refusalReason',
        'completedAt',
        'updatedAt'
      ]);
    }

    function isValidManualAssignmentEventCreation() {
      let orderId = request.resource.data.orderId;
      let agentId = request.resource.data.agentId;

      return isAdmin()
        && hasValidOrderAssignmentEventKeys()
        && request.resource.data.schemaVersion == 1
        && orderId is string
        && agentId is string
        && request.resource.data.orderReference is string
        && request.resource.data.agentName is string
        && request.resource.data.agentName.size() >= 2
        && request.resource.data.agentName.size() <= 80
        && request.resource.data.assignedByUserId == request.auth.uid
        && request.resource.data.mode == 'manual'
        && request.resource.data.status == 'assigned'
        && request.resource.data.assignedAt == request.time
        && request.resource.data.acceptedAt == null
        && request.resource.data.refusedAt == null
        && request.resource.data.refusalReason == null
        && request.resource.data.completedAt == null
        && request.resource.data.updatedAt == request.time
        && existsAfter(
          /databases/$(database)/documents/orders/$(orderId)
        )
        && getAfter(
          /databases/$(database)/documents/orders/$(orderId)
        ).data.assignedAgentId == agentId
        && getAfter(
          /databases/$(database)/documents/orders/$(orderId)
        ).data.assignedAgentName == request.resource.data.agentName
        && getAfter(
          /databases/$(database)/documents/orders/$(orderId)
        ).data.reference == request.resource.data.orderReference
        && getAfter(
          /databases/$(database)/documents/orders/$(orderId)
        ).data.assignmentStatus == 'assigned';
    }

    match /orders/{orderId} {
      allow create: if isValidCustomerOrderCreation()
        || isValidStaffOrderCreation();
      allow get: if isOrderOwner()
        || isStaff()
        || (
          isAgent()
          && resource.data.get('assignedAgentId', null) == request.auth.uid
        );
      allow list: if isStaff()
        || isOrderOwner()
        || (
          isAgent()
          && resource.data.get('assignedAgentId', null) == request.auth.uid
        );
      allow update: if isValidCustomerPaymentDeclaration()
        || isValidCustomerExpiration()
        || isValidStaffOrderUpdate()
        || isValidAdminManualAssignment();
      allow delete: if false;
    }

    match /orderAssignments/{assignmentId} {
      allow get: if isAdmin()
        || (
          isAgent()
          && resource.data.agentId == request.auth.uid
        );
      allow list: if isAdmin()
        || (
          isAgent()
          && resource.data.agentId == request.auth.uid
        );
      allow create: if isValidManualAssignmentEventCreation();
      allow update, delete: if false;
    }

    match /offers/{offerId} {
      allow get, list: if isStaff() || resource.data.isActive == true;
      allow create: if isValidOfferCreation();
      allow update: if isValidOfferUpdate();
      allow delete: if false;
    }

    match /agentProfiles/{agentId} {
      allow get: if isAdmin()
        || (isAgent() && request.auth.uid == agentId);
      allow list: if isAdmin();
      allow create: if isValidAdminAgentProfileCreation(agentId);
      allow update: if isValidAdminAgentProfileUpdate(agentId)
        || isValidAgentOperationalUpdate(agentId);
      allow delete: if false;
    }

    match /zones/{zoneId} {
      allow get, list: if isStaff() || isAgent();
      allow create: if isValidZoneCreation();
      allow update: if isValidZoneUpdate();
      allow delete: if false;
    }

    match /agentIssues/{issueId} {
      allow get: if isAdmin()
        || (
          isAgent()
          && resource.data.agentId == request.auth.uid
        );
      allow list: if isAdmin()
        || (
          isAgent()
          && resource.data.agentId == request.auth.uid
        );
      allow create: if isValidAgentIssueCreation();
      allow update: if isValidAdminAgentIssueUpdate();
      allow delete: if false;
    }

    match /users/{userId} {
      allow get: if isSignedIn()
        && (
          request.auth.uid == userId
          || isStaff()
        );
      allow list: if isStaff();
      allow create: if isValidPendingStaffProfileCreation(userId);
      allow update: if isValidAdminUserUpdate(userId);
      allow delete: if false;
    }
  }
}
